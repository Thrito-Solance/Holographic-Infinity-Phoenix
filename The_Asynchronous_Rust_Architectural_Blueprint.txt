fn bit_reverse_u64(mut n: u64) -> u64 {
    let mut rev = 0;
    let mut bits = 17; // Bounded bit-length L based on prime P
    while bits > 0 {
        rev <<= 1;
        rev |= n & 1;
        n >>= 1;
        bits -= 1;
    }
    rev % P
}

fn next_prime(mut n: u64) -> u64 {
    n += 1;
    while !is_prime(n) && n < P { n += 1; }
    n % P
}

    if n <= 1 { return false; }
    for i in 2..=((n as f64).sqrt() as u64) {
        if n % i == 0 { return false; }

    true

    }

    // Quaternionic Conjugate
    fn conjugate(self) -> Self {
        Quaternion {
            a: self.a,
            b: (P - self.b) % P,
            c: (P - self.c) % P,
            d: (P - self.d) % P,
        }
    }

    // Modular Multiplicative Norm Inverse (Fermat's Little Theorem Integration)
    fn norm_sq(self) -> u64 {
        ((self.a * self.a)%P + (self.b * self.b)%P + (self.c * self.c)%P + (self.d * self.d)%P) % P
    }

    fn inverse(self) -> Self {
        let n_sq = self.norm_sq();
        if n_sq == 0 { return Quaternion::new(0, 0, 0, 0); }
        // Inverse = q_conj * (norm_sq)^(p-2) mod p
        let inv_norm = modular_pow(n_sq, P - 2, P);
        let conj = self.conjugate();
        Quaternion {
            a: (conj.a * inv_norm) % P,
            b: (conj.b * inv_norm) % P,
            c: (conj.c * inv_norm) % P,
            d: (conj.d * inv_norm) % P,
        }
    }
}

fn modular_pow(mut base: u64, mut exp: u64, m: u64) -> u64 {
    let mut res = 1;
    base %= m;
    while exp > 0 {
        if exp % 2 == 1 { res = (res * base) % m; }
        base = (base * base) % m;
        exp /= 2;
    }
    res
}
